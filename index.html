const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); // --- 1. GIAO DIỆN CLIENT (Tích hợp trực tiếp HTML/CSS/JS) --- const htmlContent = ` Game Caro 15x15

Game Caro 15x15

Đang tải...

Kết quả

Bạn có muốn tiếp tục chơi chế độ này không?

`; // Route trả về giao diện web trực tiếp app.get('/', (req, res) => { res.send(htmlContent); }); // --- 2. XỬ LÝ SERVER LOGIC & SOCKET.IO --- const rooms = new Map(); let waitingPlayer = null; io.on('connection', (socket) => { socket.on('createRoom', (roomId) => { if (!/^\d{5}$/.test(roomId)) { return socket.emit('errorMsg', 'ID phòng phải gồm đúng 5 chữ số (0-9).'); } if (rooms.has(roomId)) { return socket.emit('errorMsg', 'ID phòng đã tồn tại! Vui lòng chọn ID khác.'); } socket.join(roomId); rooms.set(roomId, { players: [socket.id] }); socket.roomId = roomId; socket.symbol = 'X'; socket.emit('roomCreated', { roomId, symbol: 'X' }); }); socket.on('joinRoom', (roomId) => { const room = rooms.get(roomId); if (!room) return socket.emit('errorMsg', 'Phòng không tồn tại!'); if (room.players.length >= 2) return socket.emit('errorMsg', 'Phòng đã đủ 2 người chơi!'); socket.join(roomId); room.players.push(socket.id); socket.roomId = roomId; socket.symbol = 'O'; socket.emit('roomJoined', { roomId, symbol: 'O' }); io.to(roomId).emit('gameStart', { turn: 'X' }); }); socket.on('quickMatch', () => { if (waitingPlayer && waitingPlayer.id !== socket.id) { const roomId = `quick_${waitingPlayer.id}_${socket.id}`; socket.join(roomId); waitingPlayer.join(roomId); rooms.set(roomId, { players: [waitingPlayer.id, socket.id] }); waitingPlayer.roomId = roomId; waitingPlayer.symbol = 'X'; socket.roomId = roomId; socket.symbol = 'O'; waitingPlayer.emit('roomJoined', { roomId, symbol: 'X' }); socket.emit('roomJoined', { roomId, symbol: 'O' }); io.to(roomId).emit('gameStart', { turn: 'X' }); waitingPlayer = null; } else { waitingPlayer = socket; socket.emit('waitingMatch'); } }); socket.on('makeMove', ({ row, col }) => { if (socket.roomId) { socket.to(socket.roomId).emit('opponentMove', { row, col, symbol: socket.symbol }); } }); socket.on('requestRematch', () => { if (socket.roomId) { socket.to(socket.roomId).emit('rematchRequested'); } }); socket.on('disconnect', () => { if (waitingPlayer && waitingPlayer.id === socket.id) waitingPlayer = null; if (socket.roomId && rooms.has(socket.roomId)) { io.to(socket.roomId).emit('playerLeft'); rooms.delete(socket.roomId); } }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Server đang chạy tại http://localhost:${PORT}`); });
